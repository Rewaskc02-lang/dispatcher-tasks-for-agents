# dispatcher-tasks-for-agents: one `tick` every 5 minutes (add with `crontab -e`).
# Replace INSTALL_DIR with the folder of the project; PATH must reach your agent CLIs (`command -v claude` etc.).
*/5 * * * * PATH="$HOME/.local/bin:/usr/local/bin:/usr/bin:/bin" /usr/bin/python3 INSTALL_DIR/dispatcher.py tick >> "$HOME/.dispatcher-tasks.log" 2>&1
